# Stock members that an overlay DROPS and something still REFERENCES -- each traps if
# reached. Regenerate: java -cp out overlay.OverlayCheck --write test/overlay/known-gaps.txt
#
# This file is a BACKLOG, not an approval list. A line here means the gap is known and
# unfixed, not that it is safe -- most are cold paths in the JUnit jar, but the ones that
# bit us (Boolean.getBoolean, ConcurrentHashMap.<init>(IFI)V) looked exactly this cold
# until the day they ran. The check exists to stop NEW ones appearing silently.
java/lang/Class#getGenericSuperclass()Ljava/lang/reflect/Type;
java/lang/Class#getProtectionDomain()Ljava/security/ProtectionDomain;
java/lang/Class#toGenericString()Ljava/lang/String;
#
